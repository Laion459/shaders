#version 330 core

in vec3 position; //att 0 — frame atual
in vec3 cor; // att 1
in vec3 normal; // att 2
in vec2 textuv; // att 3
in vec3 position2; // att 4 — proximo frame

uniform mat4 model;
uniform mat4 view;
//uniform mat4 projection;

uniform float ang;
uniform float interpolacao;

out vec3 color;
out vec3 norm;
out vec2 uv;

void main(void){

	// Interpolacao de frames MD2 no vertex shader
	vec3 postmp = mix(position, position2, interpolacao);

	// Efeito de onda opcional do projeto base (pode zerar ang se nao quiser)
	postmp.y = postmp.y + sin(ang + postmp.z * 0.2) * 0.0;

	gl_Position = (vec4(postmp,1.0)*model*view)*vec4(1,1,0.1,1);
	
	
	norm = normal;
	uv = vec2(textuv.x, textuv.y);
	color = vec3(cor.x,cor.y,cor.z);
	
	//color = vec3((normal.x+1)*0.5,(normal.y+1)*0.5,(normal.z+1)*0.5);
	//color = vec3(textuv.x,0,textuv.y);
}
